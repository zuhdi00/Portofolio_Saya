Petunjuk Kabel RS232 (DB9)
Gunakan kabel 2-core shielded, max 10 meter.
Indikator pin 1 (TXD) → PC DB9 pin 2 (RXD)
Indikator pin 2 (GND) → PC DB9 pin 5 (GND)
Kabel > 3m: pasang magnetic loop di interface RS232.
Tidak ada DB9? Gunakan USB-to-RS232 adapter (CH340/CP2102/FTDI).
Setting di Timbangan (Menu P)
Tekan [Func] lebih dari 5 detik → masuk mode pengaturan.
P2  = Baud rate    (1=9600 / 2=4800 / 3=2400 / 4=1200)
P3  = Mode output  (1=Net / 2=Gross)
P4  = RS232 mode   (1=Stop / 2=Continuous / 3=Stable only)
Tekan [Func] untuk simpan setelah setting.

/** * Parser RS232 XK3190-A12(E) — 15-byte ASCII packet * * Byte 1 : 'W' (0x57) — header * Byte 2 : 'G'=gross / 'N'=net — weight type * Byte 3–9 : weight + decimal pt — e.g. "004.139" * Byte 10–11 : 'kg' (0x6B 0x67) * Byte 12–13 : checksum (ASCII-encoded nibbles) * Byte 14 : 0x0D (CR) * Byte 15 : 0x0A (LF) * * Overload → byte 10='O'(0x4F), 11='L'(0x4C) * Underflow → byte 10='L'(0x4C), 11='O'(0x4F) */ function parseXK3190Packet(bytes) { if (bytes.length !== 15) return null; if (bytes[0] !== 0x57) return null; if (bytes[13] !== 0x0D || bytes[14] !== 0x0A) return null; if (bytes[9] === 0x4F && bytes[10] === 0x4C) return { status: 'overload' }; if (bytes[9] === 0x4C && bytes[10] === 0x4F) return { status: 'underflow' }; const type = bytes[1] === 0x4E ? 'net' : 'gross'; const weightStr = bytes.slice(2, 9) .map(b => String.fromCharCode(b)) .join('').trim(); const weight = parseFloat(weightStr); // Checksum: sum byte 1–11, encode each nibble const sum = bytes.slice(0, 11) .reduce((a, b) => a + b, 0) & 0xFF; const hi = (sum >> 4) & 0x0F; const lo = sum & 0x0F; const encHi = hi <= 9 ? hi + 0x30 : hi + 0x37; const encLo = lo <= 9 ? lo + 0x30 : lo + 0x37; const checksumOk = bytes[11] === encHi && bytes[12] === encLo; const raw = bytes.map(b => b.toString(16) .padStart(2, '0') .toUpperCase()).join(' '); return { status: 'ok', type, weight, checksumOk, raw }; } // ── Web Serial API (Chrome/Edge 89+) ─────────────────────────── async function connectSerial(baudRate = 1200) { const port = await navigator.serial.requestPort(); await port.open({ baudRate, dataBits: 8, stopBits: 1, parity: 'even' // sesuai spesifikasi manual }); const reader = port.readable.getReader(); const buffer = []; while (true) { const { value, done } = await reader.read(); if (done) break; buffer.push(...Array.from(value)); while (buffer.length >= 15) { const idx = buffer.indexOf(0x57); if (idx === -1) { buffer.length = 0; break; } if (idx > 0) { buffer.splice(0, idx); } if (buffer.length < 15) break; const pkt = buffer.splice(0, 15); const result = parseXK3190Packet(pkt); if (result) console.log(result); } } }
